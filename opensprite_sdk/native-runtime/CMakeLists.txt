cmake_minimum_required(VERSION 3.16)
project(opensprite_robot_runtime LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "The native device gateway requires Linux")
endif()
if(NOT TARGET opensprite_dds)
  add_subdirectory(${CMAKE_CURRENT_SOURCE_DIR}/../native-dds ${CMAKE_CURRENT_BINARY_DIR}/dds)
endif()
find_package(nlohmann_json 3.9 REQUIRED)
find_package(Threads REQUIRED)
option(OPENSPRITE_WITH_MUJOCO "Build the native MuJoCo device backend" ON)
set(MUJOCO_ROOT "" CACHE PATH "MuJoCo installation directory containing include/mujoco and libmujoco.so")
if(OPENSPRITE_WITH_MUJOCO)
  find_path(MUJOCO_INCLUDE_DIR mujoco/mujoco.h HINTS ${MUJOCO_ROOT}/include)
  find_library(MUJOCO_LIBRARY NAMES mujoco libmujoco.so.3.11.0 HINTS ${MUJOCO_ROOT} ${MUJOCO_ROOT}/lib)
  if(NOT MUJOCO_INCLUDE_DIR OR NOT MUJOCO_LIBRARY)
    message(FATAL_ERROR "MuJoCo C library/headers not found. Set MUJOCO_ROOT, use tools/build_robot.sh, or configure OPENSPRITE_WITH_MUJOCO=OFF for a hardware-only runtime.")
  endif()
  set(device_sim_source src/mujoco_backend.cpp)
else()
  set(device_sim_source src/no_mujoco_backend.cpp)
endif()
add_library(opensprite_runtime STATIC src/runtime.cpp src/hardware_backend.cpp ${device_sim_source})
target_compile_options(opensprite_runtime PRIVATE -Wall -Wextra -Wpedantic)
target_include_directories(opensprite_runtime PUBLIC include PRIVATE ../native/src)
target_link_libraries(opensprite_runtime PUBLIC opensprite_dds nlohmann_json::nlohmann_json Threads::Threads)
if(OPENSPRITE_WITH_MUJOCO)
  target_include_directories(opensprite_runtime PRIVATE ${MUJOCO_INCLUDE_DIR})
  target_link_libraries(opensprite_runtime PUBLIC ${MUJOCO_LIBRARY})
endif()
add_executable(opensprite_robot src/main.cpp)
target_compile_options(opensprite_robot PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(opensprite_robot PRIVATE opensprite_runtime)
set_target_properties(opensprite_robot PROPERTIES INSTALL_RPATH "$ORIGIN/../lib")
install(TARGETS opensprite_robot RUNTIME DESTINATION bin)
if(OPENSPRITE_WITH_MUJOCO)
  get_filename_component(mujoco_real_library "${MUJOCO_LIBRARY}" REALPATH)
  install(FILES "${mujoco_real_library}" DESTINATION lib)
  install(FILES ../licenses/MuJoCo-Apache-2.0.txt ../licenses/MuJoCo-THIRD-PARTY.txt
    DESTINATION share/licenses/opensprite_robot)
endif()
