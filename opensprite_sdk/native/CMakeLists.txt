cmake_minimum_required(VERSION 3.16)
project(opensprite_sdk_native LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "The SocketCAN deployment executable requires Linux")
endif()
add_executable(sprite_can_shadow vendor/sprite_can_shadow.cpp)
target_compile_options(sprite_can_shadow PRIVATE -Wall -Wextra -Wpedantic -Werror)
target_link_libraries(sprite_can_shadow PRIVATE pthread)

# Optional ICM45686 SPI acquisition and offline estimator/packet tests.
add_subdirectory(icm45686)
enable_testing()
add_executable(test_native_contract tests/contract_test.cpp)
target_compile_options(test_native_contract PRIVATE -Wall -Wextra -Wpedantic -Werror -UNDEBUG)
target_link_libraries(test_native_contract PRIVATE pthread)
add_test(NAME native_contract COMMAND test_native_contract)
